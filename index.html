async function requestLink() { const email = document.getElementById('email').value.trim(); if (!email) return updateStatus('Please type an email address.'); const btn = document.getElementById('action-btn'); btn.innerText = 'Connecting...'; try { const res = await fetch(window.location.origin + '/api/auth/send-link', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: email }) }); const data = await res.json(); if (res.ok) { btn.innerText = 'Sent!'; updateStatus('Link Dispatched! Check your email inbox.', true); beginTracking(email); } else { btn.innerText = 'Verify Email'; updateStatus(data.error || 'Server error.'); } } catch (err) { btn.innerText = 'Verify Email'; updateStatus('Network Error: ' + err.message); } }
